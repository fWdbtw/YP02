
CREATE TABLE "Category" (
    "CategoryId" SERIAL4 NOT NULL PRIMARY KEY,
    "CategoryName" VARCHAR(100) NOT NULL,
    "ParentCategoryId" INT NULL REFERENCES "Category"("CategoryId") ON DELETE SET NULL
);

CREATE TABLE "ClientType" (
    "ClientTypeId" SERIAL4 NOT NULL PRIMARY KEY,
    "ClientTypeName" VARCHAR(50) NOT NULL,
    "Color" VARCHAR(10) NOT NULL
);

CREATE TABLE "Developer" (
    "DeveloperId" SERIAL4 NOT NULL PRIMARY KEY,
    "DeveloperName" VARCHAR(200) NOT NULL,
    "WorkBeginDate" DATE NOT NULL
);

CREATE TABLE "Season" (
    "SeasonId" SERIAL4 NOT NULL PRIMARY KEY,
    "SeasonName" VARCHAR(50) NOT NULL,
    "StartDate" DATE NOT NULL,
    "EndDate" DATE NOT NULL
);

CREATE TABLE "Service" (
    "ServiceId" SERIAL4 NOT NULL PRIMARY KEY,
    "ServiceName" VARCHAR(200) NOT NULL,
    "ServiceDescription" TEXT NULL,
    "Price" DECIMAL(12,2) NOT NULL,
    "TimeLength" INT NOT NULL,
    "MainPhoto" VARCHAR(100) NOT NULL,
    "SecondPhotos" VARCHAR(100) NULL,
    "Discount" SMALLINT NOT NULL CHECK ("Discount" BETWEEN 0 AND 100)
);

CREATE TABLE "Speciality" (
    "SpecialityId" SERIAL4 NOT NULL PRIMARY KEY,
    "SpecialityName" VARCHAR(50) NOT NULL,
    "SpecialityPay" DECIMAL(12,2) NOT NULL
);

CREATE TABLE "Client" (
    "ClientId" SERIAL4 NOT NULL PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50) NULL,
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F', 'M')),
    "Phone" VARCHAR(20) NOT NULL,
    "Email" VARCHAR(100) NOT NULL,
    "Photo" VARCHAR(50) NULL,
    "DateOfRegistration" DATE NOT NULL,
    "ClientTypeId" INT NULL REFERENCES "ClientType"("ClientTypeId") ON DELETE SET NULL
);

CREATE TABLE "Good" (
    "GoodId" SERIAL4 NOT NULL PRIMARY KEY,
    "GoodName" VARCHAR(200) NOT NULL,
    "MainPhoto" VARCHAR(50) NOT NULL,
    "SecondPhotos" VARCHAR(100) NULL,
    "Price" DECIMAL(12,2) NOT NULL,
    "Weight" FLOAT NULL,
    "Width" FLOAT NULL,
    "Height" FLOAT NULL,
    "Length" FLOAT NULL,
    "IsActive" BOOLEAN NOT NULL,
    "DeveloperId" INT NOT NULL REFERENCES "Developer"("DeveloperId") ON DELETE RESTRICT,
    "CategoryId" INT NULL REFERENCES "Category"("CategoryId") ON DELETE SET NULL
);

CREATE TABLE "Worker" (
    "WorkerId" SERIAL4 NOT NULL PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50) NULL,
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F', 'M')),
    "PassportSeries" VARCHAR(10) NOT NULL,
    "PassportId" VARCHAR(10) NOT NULL,
    "PassportCode" VARCHAR(10) NOT NULL,
    "IsActive" BOOLEAN NOT NULL,
    "PayCoeff" NUMERIC(8,3) NOT NULL,
    "SpecialityId" INT NOT NULL REFERENCES "Speciality"("SpecialityId") ON DELETE RESTRICT
);

CREATE TABLE "Complect" (
    "ComplectId" SERIAL4 NOT NULL PRIMARY KEY,
    "MainGoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE CASCADE,
    "SecondGoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE CASCADE
);

CREATE TABLE "SeasonGood" (
    "SeasonGoodId" SERIAL4 NOT NULL PRIMARY KEY,
    "SeasonId" INT NOT NULL REFERENCES "Season"("SeasonId") ON DELETE CASCADE,
    "GoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE CASCADE
);

CREATE TABLE "Sell" (
    "SellId" SERIAL4 NOT NULL PRIMARY KEY,
    "GoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE RESTRICT,
    "Count" INT NOT NULL,
    "DateSell" TIMESTAMP NOT NULL
);

CREATE TABLE "Store" (
    "StoreId" SERIAL4 NOT NULL PRIMARY KEY,
    "GoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE CASCADE,
    "Count" INT NOT NULL
);

CREATE TABLE "TimeSheet" (
    "TimeSheetId" SERIAL4 NOT NULL PRIMARY KEY,
    "WorkerId" INT NOT NULL REFERENCES "Worker"("WorkerId") ON DELETE CASCADE,
    "ServiceId" INT NOT NULL REFERENCES "Service"("ServiceId") ON DELETE CASCADE,
    "ServiceTime" TIME NOT NULL,
    "GenderCheck" BOOLEAN NOT NULL
);

CREATE TABLE "Visit" (
    "VisitId" SERIAL4 NOT NULL PRIMARY KEY,
    "ClientId" INT NOT NULL REFERENCES "Client"("ClientId") ON DELETE RESTRICT,
    "ServiceId" INT NOT NULL REFERENCES "Service"("ServiceId") ON DELETE RESTRICT,
    "DateRegs" TIMESTAMP NULL,
    "DateActual" TIMESTAMP NOT NULL,
    "Paid" BOOLEAN NULL,
    "Review" TEXT NULL,
    "Documents" TEXT NULL
);

CREATE TABLE "SellVisit" (
    "SellVisitId" SERIAL4 NOT NULL PRIMARY KEY,
    "SellId" INT NOT NULL REFERENCES "Sell"("SellId") ON DELETE CASCADE,
    "VisitId" INT NOT NULL REFERENCES "Visit"("VisitId") ON DELETE CASCADE
);
