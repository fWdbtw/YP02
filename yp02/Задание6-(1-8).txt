CREATE TABLE "ClientType" (
    "ClientTypeId" SERIAL4 PRIMARY KEY,
    "ClientTypeName" VARCHAR(50) NOT NULL,
    "DiscountPercent" NUMERIC(5,2) NOT NULL CHECK ("DiscountPercent" BETWEEN 0 AND 100)
);

CREATE TABLE "Client" (
    "ClientId" SERIAL4 PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50),
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F','M')),
    "Phone" VARCHAR(20) NOT NULL,
    "Email" VARCHAR(100) NOT NULL,
    "Address" VARCHAR(200),
    "DateOfRegistration" DATE NOT NULL,
    "ClientTypeId" INT REFERENCES "ClientType"("ClientTypeId") ON DELETE SET NULL
);

CREATE TABLE "Car" (
    "CarId" SERIAL4 PRIMARY KEY,
    "ClientId" INT NOT NULL REFERENCES "Client"("ClientId") ON DELETE RESTRICT,
    "Brand" VARCHAR(50) NOT NULL,
    "Model" VARCHAR(50) NOT NULL,
    "Year" INT NOT NULL,
    "VIN" VARCHAR(17) NOT NULL CHECK (char_length("VIN") = 17),
    "LicensePlate" VARCHAR(20) NOT NULL,
    "Color" VARCHAR(30),
    "Mileage" INT CHECK ("Mileage" >= 0)
);

CREATE TABLE "Speciality" (
    "SpecialityId" SERIAL4 PRIMARY KEY,
    "SpecialityName" VARCHAR(50) NOT NULL,
    "HourlyRate" NUMERIC(12,2) NOT NULL CHECK ("HourlyRate" >= 0)
);

CREATE TABLE "Worker" (
    "WorkerId" SERIAL4 PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50),
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F','M')),
    "PassportSeries" VARCHAR(10) NOT NULL,
    "PassportNumber" VARCHAR(10) NOT NULL,
    "PassportCode" VARCHAR(10) NOT NULL,
    "Phone" VARCHAR(20) NOT NULL,
    "IsActive" BOOLEAN NOT NULL,
    "PayCoeff" NUMERIC(8,2) NOT NULL CHECK ("PayCoeff" >= 0),
    "SpecialityId" INT NOT NULL REFERENCES "Speciality"("SpecialityId") ON DELETE RESTRICT
);

CREATE TABLE "Developer" (
    "DeveloperId" SERIAL4 PRIMARY KEY,
    "DeveloperName" VARCHAR(200) NOT NULL,
    "Country" VARCHAR(100),
    "ContactPhone" VARCHAR(20)
);

CREATE TABLE "Category" (
    "CategoryId" SERIAL4 PRIMARY KEY,
    "CategoryName" VARCHAR(100) NOT NULL,
    "ParentCategoryId" INT REFERENCES "Category"("CategoryId") ON DELETE SET NULL,
    CHECK ("ParentCategoryId" IS DISTINCT FROM "CategoryId")
);

CREATE TABLE "Good" (
    "GoodId" SERIAL4 PRIMARY KEY,
    "GoodName" VARCHAR(200) NOT NULL,
    "ArticleNumber" VARCHAR(50) NOT NULL,
    "Price" NUMERIC(12,2) NOT NULL CHECK ("Price" >= 0),
    "Weight" NUMERIC(10,2) CHECK ("Weight" >= 0),
    "IsActive" BOOLEAN NOT NULL,
    "DeveloperId" INT NOT NULL REFERENCES "Developer"("DeveloperId") ON DELETE RESTRICT,
    "CategoryId" INT REFERENCES "Category"("CategoryId") ON DELETE SET NULL
);

CREATE TABLE "Store" (
    "StoreId" SERIAL4 PRIMARY KEY,
    "GoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE CASCADE,
    "Quantity" INT NOT NULL CHECK ("Quantity" >= 0),
    "MinQuantity" INT NOT NULL CHECK ("MinQuantity" >= 0)
);

CREATE TABLE "Service" (
    "ServiceId" SERIAL4 PRIMARY KEY,
    "ServiceName" VARCHAR(200) NOT NULL,
    "ServiceDescription" TEXT,
    "Price" NUMERIC(12,2) NOT NULL CHECK ("Price" >= 0),
    "TimeLength" INT NOT NULL CHECK ("TimeLength" > 0),
    "IsActive" BOOLEAN NOT NULL
);

CREATE TABLE "TimeSheet" (
    "TimeSheetId" SERIAL4 PRIMARY KEY,
    "WorkerId" INT NOT NULL REFERENCES "Worker"("WorkerId") ON DELETE RESTRICT,
    "ServiceId" INT NOT NULL REFERENCES "Service"("ServiceId") ON DELETE RESTRICT,
    "WorkDate" DATE NOT NULL,
    "StartTime" TIME NOT NULL,
    "EndTime" TIME NOT NULL CHECK ("EndTime" > "StartTime")
);

CREATE TABLE "Visit" (
    "VisitId" SERIAL4 PRIMARY KEY,
    "ClientId" INT NOT NULL REFERENCES "Client"("ClientId") ON DELETE RESTRICT,
    "CarId" INT NOT NULL REFERENCES "Car"("CarId") ON DELETE RESTRICT,
    "ServiceId" INT NOT NULL REFERENCES "Service"("ServiceId") ON DELETE RESTRICT,
    "WorkerId" INT NOT NULL REFERENCES "Worker"("WorkerId") ON DELETE RESTRICT,
    "DateCreated" TIMESTAMP NOT NULL,
    "DateActual" TIMESTAMP,
    "Status" VARCHAR(50) NOT NULL,
    "IsPaid" BOOLEAN NOT NULL,
    "TotalAmount" NUMERIC(12,2) NOT NULL CHECK ("TotalAmount" >= 0),
    "Notes" TEXT
);

CREATE TABLE "Sell" (
    "SellId" SERIAL4 PRIMARY KEY,
    "GoodId" INT NOT NULL REFERENCES "Good"("GoodId") ON DELETE RESTRICT,
    "Quantity" INT NOT NULL CHECK ("Quantity" >= 0),
    "DateSell" TIMESTAMP NOT NULL,
    "TotalPrice" NUMERIC(12,2) NOT NULL CHECK ("TotalPrice" >= 0)
);

CREATE TABLE "SellVisit" (
    "SellVisitId" SERIAL4 PRIMARY KEY,
    "SellId" INT NOT NULL REFERENCES "Sell"("SellId") ON DELETE CASCADE,
    "VisitId" INT NOT NULL REFERENCES "Visit"("VisitId") ON DELETE CASCADE,
    UNIQUE ("SellId", "VisitId")
);

CREATE TABLE "Season" (
    "SeasonId" SERIAL4 PRIMARY KEY,
    "SeasonName" VARCHAR(50) NOT NULL,
    "StartDate" DATE NOT NULL,
    "EndDate" DATE NOT NULL CHECK ("EndDate" >= "StartDate")
);

CREATE TABLE "SeasonService" (
    "SeasonServiceId" SERIAL4 PRIMARY KEY,
    "SeasonId" INT NOT NULL REFERENCES "Season"("SeasonId") ON DELETE CASCADE,
    "ServiceId" INT NOT NULL REFERENCES "Service"("ServiceId") ON DELETE CASCADE,
    UNIQUE ("SeasonId", "ServiceId")
);
