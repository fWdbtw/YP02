CREATE TABLE "InsuranceType" (
    "InsuranceTypeId" SERIAL4 PRIMARY KEY,
    "InsuranceName" VARCHAR(100) NOT NULL,
    "DiscountPercent" NUMERIC(5,2) NOT NULL CHECK ("DiscountPercent" BETWEEN 0 AND 100)
);

CREATE TABLE "Patient" (
    "PatientId" SERIAL4 PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50),
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F','M')),
    "Phone" VARCHAR(20) NOT NULL,
    "Email" VARCHAR(100),
    "DateOfRegistration" DATE NOT NULL,
    "InsuranceTypeId" INT REFERENCES "InsuranceType"("InsuranceTypeId") ON DELETE SET NULL
);

CREATE TABLE "Specialization" (
    "SpecializationId" SERIAL4 PRIMARY KEY,
    "SpecializationName" VARCHAR(100) NOT NULL,
    "ConsultationPrice" NUMERIC(12,2) NOT NULL CHECK ("ConsultationPrice" >= 0)
);

CREATE TABLE "Doctor" (
    "DoctorId" SERIAL4 PRIMARY KEY,
    "LastName" VARCHAR(50) NOT NULL,
    "FirstName" VARCHAR(50) NOT NULL,
    "MiddleName" VARCHAR(50),
    "DateOfBirth" DATE NOT NULL,
    "Gender" CHAR(1) NOT NULL CHECK ("Gender" IN ('F','M')),
    "PassportSeries" VARCHAR(10) NOT NULL,
    "PassportNumber" VARCHAR(10) NOT NULL,
    "IsActive" BOOLEAN NOT NULL,
    "PayCoeff" NUMERIC(8,2) NOT NULL CHECK ("PayCoeff" >= 0),
    "SpecializationId" INT NOT NULL REFERENCES "Specialization"("SpecializationId") ON DELETE RESTRICT
);

CREATE TABLE "Department" (
    "DepartmentId" SERIAL4 PRIMARY KEY,
    "DepartmentName" VARCHAR(100) NOT NULL,
    "FloorNumber" INT NOT NULL CHECK ("FloorNumber" > 0)
);

CREATE TABLE "MedicalService" (
    "ServiceId" SERIAL4 PRIMARY KEY,
    "ServiceName" VARCHAR(200) NOT NULL,
    "ServiceDescription" TEXT,
    "Price" NUMERIC(12,2) NOT NULL CHECK ("Price" >= 0),
    "DurationMinutes" INT NOT NULL CHECK ("DurationMinutes" > 0),
    "MainPhoto" VARCHAR(100) NOT NULL,
    "DepartmentId" INT NOT NULL REFERENCES "Department"("DepartmentId") ON DELETE RESTRICT
);

CREATE TABLE "Appointment" (
    "AppointmentId" SERIAL4 PRIMARY KEY,
    "PatientId" INT NOT NULL REFERENCES "Patient"("PatientId") ON DELETE RESTRICT,
    "DoctorId" INT NOT NULL REFERENCES "Doctor"("DoctorId") ON DELETE RESTRICT,
    "ServiceId" INT NOT NULL REFERENCES "MedicalService"("ServiceId") ON DELETE RESTRICT,
    "AppointmentDate" TIMESTAMP NOT NULL,
    "ActualDate" TIMESTAMP,
    "IsPaid" BOOLEAN,
    "Status" VARCHAR(50) NOT NULL,
    "DoctorNotes" TEXT
);

CREATE TABLE "Manufacturer" (
    "ManufacturerId" SERIAL4 PRIMARY KEY,
    "ManufacturerName" VARCHAR(200) NOT NULL,
    "Country" VARCHAR(100) NOT NULL
);

CREATE TABLE "Medicine" (
    "MedicineId" SERIAL4 PRIMARY KEY,
    "MedicineName" VARCHAR(200) NOT NULL,
    "Price" NUMERIC(12,2) NOT NULL CHECK ("Price" >= 0),
    "IsActive" BOOLEAN NOT NULL,
    "ManufacturerId" INT NOT NULL REFERENCES "Manufacturer"("ManufacturerId") ON DELETE RESTRICT
);

CREATE TABLE "PharmacyStock" (
    "StockId" SERIAL4 PRIMARY KEY,
    "MedicineId" INT NOT NULL REFERENCES "Medicine"("MedicineId") ON DELETE CASCADE,
    "Quantity" INT NOT NULL CHECK ("Quantity" >= 0),
    "ExpirationDate" DATE NOT NULL
);

CREATE TABLE "Prescription" (
    "PrescriptionId" SERIAL4 PRIMARY KEY,
    "AppointmentId" INT NOT NULL REFERENCES "Appointment"("AppointmentId") ON DELETE RESTRICT,
    "MedicineId" INT NOT NULL REFERENCES "Medicine"("MedicineId") ON DELETE RESTRICT,
    "Quantity" INT NOT NULL CHECK ("Quantity" >= 0),
    "SaleDate" TIMESTAMP NOT NULL
);
